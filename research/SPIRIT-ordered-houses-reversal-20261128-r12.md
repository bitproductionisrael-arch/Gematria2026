# ספר SPIRIT — פרק 06: בתים מסודרים ופנים ואחור
28.11.2026. מקור ראשוני שנקרא: https://he.wikisource.org/wiki/ספר_יצירה פרק ד משנה יב: "שלש בונות ששה בתים. ארבעה בונות ארבעה ועשרים בתים"; פרק ב משנה ז: "חזר גלגל פנים ואחור". המקור סופר תמורות מסודרות, בניגוד לתת-קבוצות בפרק 05. ההרחבה באמצעות חבורת קליין מודרנית.
N=2k; A(x)=N-1-x, B(x)=(x+k)%N, C=AB, בהתאם לפרק 04 ולמחשבון. על שלשות מסודרות של אותיות שונות פועלת החבורה G, וכן R המהפכת סדר מיקומים. R מתחלפת עם G. כל פעולה לא טריוויאלית ב-G מקבעת לכל היותר שתי אותיות, ולכן אינה מקבעת שלשה מסודרת. מספר מחלקות G הוא N(N-1)(N-2)/4.
פעולה gR מקבעת (a,b,c) בדיוק כאשר g(b)=b ו-g(a)=c, g(c)=a. רק C כאשר k אי זוגי מאפשרת זאת: שתי בחירות לאות אמצעית ו-2(k-1) בחירות לקצה הראשון, כלומר 4(k-1). לכן מספר מחלקות תחת G×R הוא [N(N-1)(N-2)+4(k-1)*epsilon]/8, epsilon=1 כאשר k אי זוגי, אחרת 0.

for(const n of [22,24,26,28]){
 const k=n/2,f=[x=>x,x=>n-1-x,x=>(x+k)%n,x=>(n-1-x+k)%n];
 let fixed=[0,0,0,0],rev=[0,0,0,0];
 for(let a=0;a<n;a++)for(let b=0;b<n;b++)for(let c=0;c<n;c++)if(new Set([a,b,c]).size===3){
 let s=[a,b,c];f.forEach((g,i)=>{if(s.every((x,j)=>g(x)===s[j]))fixed[i]++;if(s.every((x,j)=>g(x)===s[2-j]))rev[i]++});
 }
 console.log(n,fixed,rev,fixed.reduce((a,b)=>a+b)/4,[...fixed,...rev].reduce((a,b)=>a+b)/8);
}
