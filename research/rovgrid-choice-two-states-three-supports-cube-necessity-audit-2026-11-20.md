## אימות ניתן להרצה (Python 3)
```python
from itertools import combinations, product
supports = [s for n in (1,2) for s in combinations((0,1),n)]
vertices = list(product((0,1), repeat=len(supports)))
edges = [(a,b) for a,b in combinations(vertices,2) if sum(x!=y for x,y in zip(a,b))==1]
assert (len(supports),len(vertices),len(edges)) == (3,8,12)
cells = list(product('01*', repeat=3))
stages = [{c for c in cells if c.count('*') <= k} for k in range(3)]
pairs = list(combinations(stages,2))
assert [len(a)*len(b) for a,b in pairs] == [160,208,520]
assert [sum(x!=y for x in a for y in b) for a,b in pairs] == [152,200,500]
```
